        class PRNG {
            constructor(seedStr) {
                this.seedStr = seedStr;
                this.seed = this.cyrb128(seedStr);
            }
            cyrb128(str) {
                let h1 = 1779033703, h2 = 3144134277, h3 = 1013904242, h4 = 2773480762;
                for (let i = 0, k; i < str.length; i++) {
                    k = str.charCodeAt(i);
                    h1 = h2 ^ Math.imul(h1 ^ k, 597399067);
                    h2 = h3 ^ Math.imul(h2 ^ k, 2869860233);
                    h3 = h4 ^ Math.imul(h3 ^ k, 951274213);
                    h4 = h1 ^ Math.imul(h4 ^ k, 2716044179);
                }
                h1 = Math.imul(h3 ^ (h1 >>> 18), 597399067);
                h2 = Math.imul(h4 ^ (h2 >>> 22), 2869860233);
                h3 = Math.imul(h1 ^ (h3 >>> 17), 2716044179);
                h4 = Math.imul(h2 ^ (h4 >>> 19), 2869860233);
                return (h1^h2^h3^h4) >>> 0;
            }
            next() {
                let t = this.seed += 0x6D2B79F5;
                t = Math.imul(t ^ t >>> 15, t | 1);
                t ^= t + Math.imul(t ^ t >>> 7, t | 61);
                return ((t ^ t >>> 14) >>> 0) / 4294967296;
            }
            range(min, max) { return this.next() * (max - min) + min; }
        }

        class Vector {
            constructor(x, y) { this.x = x; this.y = y; }
            add(v) { this.x += v.x; this.y += v.y; return this; }
            sub(v) { this.x -= v.x; this.y -= v.y; return this; }
            mult(n) { this.x *= n; this.y *= n; return this; }
            div(n) { this.x /= n; this.y /= n; return this; }
            magSq() { return this.x * this.x + this.y * this.y; }
            mag() { return Math.sqrt(this.magSq()); }
            normalize() { let m = this.mag(); if (m > 0) this.div(m); return this; }
            limit(max) { if (this.magSq() > max * max) this.normalize().mult(max); return this; }
            copy() { return new Vector(this.x, this.y); }
            static sub(a, b) { return new Vector(a.x - b.x, a.y - b.y); }
            static distSq(a, b) { return Vector.sub(a, b).magSq(); }
        }

        function generateLineageName(prng) {
            const a = ['Ka','Lo','Mi','Ra','Sa','Te','Vi','Zu','Ne','Po','Qi','Da'];
            const b = ['ri','na','to','va','li','se','ko','mu','ra','xi','do','pe'];
            return a[Math.floor(prng.range(0, a.length))] + b[Math.floor(prng.range(0, b.length))];
        }

        /***********************************************************************
         * SPATIAL GRID
         ***********************************************************************/
        class SpatialGrid {
            constructor(width, height, cellSize) {
                this.width = width;
                this.height = height;
                this.cellSize = cellSize;
                this.cols = Math.ceil(width / cellSize);
                this.rows = Math.ceil(height / cellSize);
                this.cells = Array.from({ length: this.cols * this.rows }, () => []);
            }
            _idx(c, r) { return r * this.cols + c; }
            clear() { for (let cell of this.cells) cell.length = 0; }
            insert(ent) {
                let c = Math.floor(ent.pos.x / this.cellSize);
                let r = Math.floor(ent.pos.y / this.cellSize);
                if (c < 0) c = 0; if (c >= this.cols) c = this.cols - 1;
                if (r < 0) r = 0; if (r >= this.rows) r = this.rows - 1;
                this.cells[this._idx(c, r)].push(ent);
            }
            getNearby(pos, radius) {
                const results = [];
                const rCells = Math.ceil(radius / this.cellSize);
                let baseCol = Math.floor(pos.x / this.cellSize);
                let baseRow = Math.floor(pos.y / this.cellSize);
                const rSq = radius * radius;
                for (let i = -rCells; i <= rCells; i++) {
                    for (let j = -rCells; j <= rCells; j++) {
                        let c = (baseCol + i) % this.cols; if (c < 0) c += this.cols;
                        let r = (baseRow + j) % this.rows; if (r < 0) r += this.rows;
                        
                        const cell = this.cells[this._idx(c, r)];
                        for (let k = 0; k < cell.length; k++) {
                            const ent = cell[k];
                            // Quick wrap-aware distance check
                            let dx = Math.abs(pos.x - ent.pos.x);
                            let dy = Math.abs(pos.y - ent.pos.y);
                            if (dx > this.width / 2) dx = this.width - dx;
                            if (dy > this.height / 2) dy = this.height - dy;
                            if (dx*dx + dy*dy <= rSq) results.push(ent);
                        }
                    }
                }
                return results;
            }
        }

        const ENTITY_PLANT = 0;

        class Resource {
            constructor(x, y, type, energy) {
                this.type = type; // 0 = Plant, 1 = Meat
                this.pos = new Vector(x, y);
                this.energy = energy;
                this.active = true;
            }
            render(ctx) {

                ctx.fillStyle = this.type === ENTITY_PLANT ? '#4ade80' : '#f87171';
                ctx.beginPath();
                const radius = this.type === ENTITY_PLANT ? Math.min(this.energy / 15, 3) : Math.min(this.energy / 30, 4);
                ctx.arc(this.pos.x, this.pos.y, radius, 0, Math.PI * 2);
                ctx.fill();
            }
        }

        class Agent {
            constructor(x, y, dna, lineage, gen) {
                this.type = ENTITY_AGENT;
                this.pos = new Vector(x, y);
                this.vel = new Vector(sim.prng.range(-1, 1), sim.prng.range(-1, 1)).normalize();
                this.acc = new Vector(0, 0);
                
                this.age = 0;
                this.generation = gen || 1;
                this.lineage = lineage || { name: generateLineageName(sim.prng), color: `hsl(${sim.prng.range(0,360)}, 70%, 60%)` };
                this.active = true;
                
                // --- GENETICS ---
                this.dna = dna || {
                    size: sim.prng.range(3, 7),           // 3(tiny) to 7(huge). Affects max energy and base metabolism.
                    speed: sim.prng.range(1.5, 3.5),      // 1.5(slow) to 3.5(fast). Affects movement energy cost.
                    perception: sim.prng.range(50, 120),  // Sight radius. Costly to maintain large brain.
                    diet: sim.prng.range(0, 1),           // 0=Strict Herbivore, 1=Strict Carnivore.
                    sociability: sim.prng.range(-0.5, 1.5), // <0 avoid, >0 flock.
                    exploration: sim.prng.range(0.1, 0.9),  // Chance to wander randomly vs strict seek.
                    mutationRate: 0.1
                };

                // Apply constraints
                this.dna.size = Math.max(2, Math.min(this.dna.size, 10));
                this.dna.speed = Math.max(1, Math.min(this.dna.speed, 5));
                this.dna.perception = Math.max(30, Math.min(this.dna.perception, 200));
                this.dna.diet = Math.max(0, Math.min(this.dna.diet, 1));
                
                // Derived Stats
                this.maxEnergy = this.dna.size * 100;
                this.energy = this.maxEnergy * 0.5; // Start at 50%
                this.reproduceThreshold = this.maxEnergy * 0.85;
                
                // State
                this.state = "Exploring";
                this.target = null;
                this.activeLinks = []; // For rendering interaction lines
            }

            mutate() {
                const rate = this.dna.mutationRate;
                const amt = 0.15; // Max 15% swing
                for (let key in this.dna) {
                    if (key !== 'mutationRate' && sim.prng.next() < rate) {
                        this.dna[key] *= (1 + sim.prng.range(-amt, amt));
                    }
                }
            }

            // Wrapping distance logic
            wrapSteer(targetPos) {
                let dx = targetPos.x - this.pos.x;
                let dy = targetPos.y - this.pos.y;
                if (Math.abs(dx) > sim.width / 2) dx = dx > 0 ? dx - sim.width : dx + sim.width;
                if (Math.abs(dy) > sim.height / 2) dy = dy > 0 ? dy - sim.height : dy + sim.height;
                let desired = new Vector(dx, dy);
                if(desired.magSq() > 0) desired.normalize().mult(this.dna.speed);
                let steer = Vector.sub(desired, this.vel);
                steer.limit(0.1); // Max force
                return steer;
            }

            behavior(nearby) {
                this.activeLinks.length = 0;
                let sep = new Vector(0, 0);
                let ali = new Vector(0, 0);
                let coh = new Vector(0, 0);
                let sepCount = 0;
                let flockCount = 0;
                let closestFood = null;
                let closestDistSq = Infinity;

                for (let ent of nearby) {
                    if (ent === this) continue;
                    const dx = Math.abs(ent.pos.x - this.pos.x);
                    const dy = Math.abs(ent.pos.y - this.pos.y);
                    const wrapDx = dx > sim.width/2 ? sim.width - dx : dx;
                    const wrapDy = dy > sim.height/2 ? sim.height - dy : dy;
                    const dSq = wrapDx*wrapDx + wrapDy*wrapDy;

                    if (ent.type === ENTITY_AGENT) {
                        if (dSq < 900) {
                            let diff = Vector.sub(this.pos, ent.pos);
                            if (Math.abs(diff.x) > sim.width/2) diff.x = diff.x > 0 ? diff.x - sim.width : diff.x + sim.width;
                            if (Math.abs(diff.y) > sim.height/2) diff.y = diff.y > 0 ? diff.y - sim.height : diff.y + sim.height;
                            if (dSq > 1) { diff.normalize().div(Math.sqrt(dSq)); sep.add(diff); sepCount++; }
                        }
                        if (dSq < 2500) {
                            ali.add(ent.vel);
                            coh.add(ent.pos);
                            flockCount++;
                        }
                    } else if (ent.type === ENTITY_PLANT || ent.type === ENTITY_MEAT) {
                        // Diet preference: low diet values seek plants, high values seek meat
                        const canEat = ent.type === ENTITY_PLANT ? this.dna.diet < 0.95 : this.dna.diet > 0.05;
                        if (canEat && dSq < closestDistSq) {
                            closestFood = ent;
                            closestDistSq = dSq;
                        }
                    }
                }

                // Apply Flocking
                if (sepCount > 0) sep.div(sepCount).normalize().mult(this.dna.speed).sub(this.vel).limit(0.15);
                if (flockCount > 0 && this.dna.sociability > 0) {
                    ali.div(flockCount).normalize().mult(this.dna.speed).sub(this.vel).limit(0.05);
                    coh.div(flockCount).normalize().mult(this.dna.speed).sub(this.vel).limit(0.05);
                    this.acc.add(ali.mult(this.dna.sociability));
                    this.acc.add(coh.mult(this.dna.sociability));
                }
                this.acc.add(sep.mult(2.0)); // Strong base separation

                // Foraging vs Exploration
                this.target = null;
                if (closestFood && sim.prng.next() > (this.dna.exploration * 0.5)) {
                    // Exploit: Go for food
                    this.state = "Foraging";
                    this.target = closestFood.pos;
                    if (sim.config.drawLinks && this === sim.inspectedAgent) this.activeLinks.push({p: closestFood.pos, type: 'food'});
                    let seekForce = this.wrapSteer(closestFood.pos);
                    this.acc.add(seekForce.mult(1.5));
                } else {
                    // Explore: Wander
                    this.state = "Exploring";
                    let wanderAngle = sim.prng.range(-0.5, 0.5);
                    let wanderForce = this.vel.copy().normalize();
                    let cos = Math.cos(wanderAngle), sin = Math.sin(wanderAngle);
                    let nx = wanderForce.x * cos - wanderForce.y * sin;
                    let ny = wanderForce.x * sin + wanderForce.y * cos;
                    wanderForce.x = nx; wanderForce.y = ny;
                    wanderForce.mult(this.dna.speed).sub(this.vel).limit(0.05);
                    this.acc.add(wanderForce.mult(this.dna.exploration));
                }
            }

            update() {
                // Physics
                this.vel.add(this.acc);
                this.vel.limit(this.dna.speed);
                this.pos.add(this.vel);
                this.acc.mult(0);

                // Screen Wrap
                if (this.pos.x < 0) this.pos.x += sim.width;
                if (this.pos.x > sim.width) this.pos.x -= sim.width;
                if (this.pos.y < 0) this.pos.y += sim.height;
                if (this.pos.y > sim.height) this.pos.y -= sim.height;

                // Energy Drain Calculation (The Trade-offs)
                // 1. Base metabolism scales with volume (size^2 roughly) and brain size (perception).
                let baseDrain = (this.dna.size * this.dna.size * 0.005) + (this.dna.perception * 0.001);
                // 2. Kinetic drain scales with current speed and mass.
                let kineticDrain = this.vel.magSq() * this.dna.size * 0.01;
                // 3. Environmental harshness (edges of map cost more)
                let centerDistX = Math.abs(this.pos.x - sim.width/2) / (sim.width/2);
                let centerDistY = Math.abs(this.pos.y - sim.height/2) / (sim.height/2);
                let harshness = Math.max(centerDistX, centerDistY); // 0 at center, 1 at edges
                let envDrain = harshness * 0.5;

                this.energy -= (baseDrain + kineticDrain + envDrain);
                this.age++;

                if (this.energy <= 0 || this.age > 3000) this.active = false;
            }

            eat(foodList) {
                // Check immediate vicinity for food
                for (let i = foodList.length - 1; i >= 0; i--) {
                    let food = foodList[i];
                    if (!food.active) continue;
                    
                    let dx = Math.abs(food.pos.x - this.pos.x);
                    let dy = Math.abs(food.pos.y - this.pos.y);
                    if (dx > sim.width/2) dx = sim.width - dx;
                    if (dy > sim.height/2) dy = sim.height - dy;
                    
                    if (dx*dx + dy*dy < (this.dna.size*this.dna.size + 16)) {
                        // Calculate digestion efficiency
                        let efficiency = 0;
                        if (food.type === ENTITY_PLANT) efficiency = 1 - this.dna.diet;
                        else efficiency = this.dna.diet;

                        // Omnivore penalty: Generalists are less efficient than specialists
                        // Max efficiency for specialist = 1. Max for strict omnivore (0.5 diet) = 0.5.
                        let digested = food.energy * efficiency;
                        
                        if (digested > 1) { // Only eat if we can digest it meaningfully
                            this.energy = Math.min(this.maxEnergy, this.energy + digested);
                            food.active = false; // Mark for removal
                        }
                    }
                }
            }

            render(ctx) {
                const isSelected = (this === sim.inspectedAgent);
                
                // Draw perception radius if selected
                if (isSelected) {
                    ctx.beginPath(); ctx.arc(this.pos.x, this.pos.y, this.dna.perception, 0, Math.PI*2);
                    ctx.fillStyle = 'rgba(255,255,255,0.05)'; ctx.fill();
                    ctx.strokeStyle = 'rgba(255,255,255,0.2)'; ctx.lineWidth = 1; ctx.stroke();
                    
                    // Draw links
                    if (sim.config.drawLinks) {
                        ctx.lineWidth = 1;
                        for(let link of this.activeLinks) {
                            ctx.beginPath(); ctx.moveTo(this.pos.x, this.pos.y);
                            // Avoid drawing lines across wrap
                            if(Vector.distSq(this.pos, link.p) < 40000) {
                                ctx.lineTo(link.p.x, link.p.y);
                                ctx.strokeStyle = link.type === 'food' ? 'rgba(56, 189, 248, 0.5)' : 'rgba(255, 255, 255, 0.2)';
                                ctx.stroke();
                            }
                        }
                    }
                }

                ctx.save();
                ctx.translate(this.pos.x, this.pos.y);
                ctx.rotate(Math.atan2(this.vel.y, this.vel.x) + Math.PI / 2);

                // Appearance based on Genetics
                // Diet determines color: 0 (Herbivore, Green 120deg), 1 (Carnivore, Red 0deg)
                let hue = (1 - this.dna.diet) * 120;
                
                // Lineage override
                if (sim.config.drawLineage) {
                    ctx.fillStyle = this.lineage.color;
                } else {
                    let lightness = 40 + (this.energy / this.maxEnergy) * 30; // Brighter when full
                    ctx.fillStyle = `hsl(${hue}, 80%, ${lightness}%)`;
                }

                // Alpha pulses if starving
                if (this.energy < this.maxEnergy * 0.2) ctx.globalAlpha = 0.5 + Math.sin(sim.ticks * 0.2) * 0.3;
                
                let s = this.dna.size;
                ctx.beginPath();
                ctx.moveTo(0, -s * 2);
                ctx.lineTo(-s, s);
                ctx.lineTo(s, s);
                ctx.closePath();
                ctx.fill();
                
                // Highlight stroke if selected
                if (isSelected) {
                    ctx.strokeStyle = '#fff'; ctx.lineWidth = 2; ctx.stroke();
                }

                ctx.restore();
            }
        }

        /***********************************************************************
         * SIMULATION ENGINE
         ***********************************************************************/
        class Simulation {
            constructor() {
                this.canvas = document.getElementById('sim-canvas');
                this.ctx = this.canvas.getContext('2d', { alpha: false });
                this.width = window.innerWidth;
                this.height = window.innerHeight;
                this.canvas.width = this.width;
                this.canvas.height = this.height;

                window.addEventListener('resize', () => {
                    this.width = window.innerWidth;
                    this.height = window.innerHeight;
                    this.canvas.width = this.width;
                    this.canvas.height = this.height;
                    if(this.grid) this.grid = new SpatialGrid(this.width, this.height, 100);
                });

                // Interaction
                this.canvas.addEventListener('mousedown', (e) => this.handleClick(e));
                this.canvas.addEventListener('touchstart', (e) => {
                    if(e.touches.length > 0) this.handleClick(e.touches[0]);
                });

                this.config = {
                    drawEnv: false, drawLinks: false, drawLineage: false, speed: 1,
                    initialPop: 250, plantSpawnRate: 4
                };

                this.isRunning = false;
                this.inspectedAgent = null;
                
                this.bindUI();
            }

            init(seedStr) {
                this.prng = new PRNG(seedStr);
                document.getElementById('inp-seed').value = seedStr;
                
                this.agents = [];
                this.resources = [];
                this.grid = new SpatialGrid(this.width, this.height, 100);
                
                this.ticks = 0;
                this.stats = { births: 0, deaths: 0, maxGen: 1 };
                this.inspectedAgent = null;
                this.updateInspectorUI();

                // Initial Pop
                for (let i = 0; i < this.config.initialPop; i++) {
                    this.agents.push(new Agent(this.prng.range(0, this.width), this.prng.range(0, this.height)));
                }
                
                // Initial Food
                for (let i=0; i< 400; i++) this.spawnPlant();

                if (!this.isRunning) {
                    this.isRunning = true;
                    document.getElementById('btn-pause').innerText = "Pause";
                    this.loop();
                }
            }

            spawnPlant() {
                // Prefer center, create a bell curve distribution
                let u = 0, v = 0;
                while(u === 0) u = this.prng.next(); //Converting [0,1) to (0,1)
                while(v === 0) v = this.prng.next();
                let randStdNormal = Math.sqrt( -2.0 * Math.log( u ) ) * Math.cos( 2.0 * Math.PI * v );
                
                // Map to screen coordinates (center concentrated)
                let x = (this.width / 2) + (randStdNormal * this.width * 0.25);
                let y = (this.height / 2) + ((Math.sqrt( -2.0 * Math.log( this.prng.next() || 0.1 ) ) * Math.cos( 2.0 * Math.PI * this.prng.next() )) * this.height * 0.25);
                
                // Wrap
                x = (x % this.width + this.width) % this.width;
                y = (y % this.height + this.height) % this.height;

                this.resources.push(new Resource(x, y, ENTITY_PLANT, this.prng.range(20, 45)));
            }

            step() {
                this.ticks++;

                // Environment Regeneration
                for(let i=0; i<this.config.plantSpawnRate; i++) {
                    if (this.resources.length < 1500) this.spawnPlant();
                }

                // Build Grid
                this.grid.clear();
                for (let a of this.agents) this.grid.insert(a);
                for (let r of this.resources) this.grid.insert(r);

                let nextAgents = [];
                let plantCount = 0, meatCount = 0;
                let dietSum = 0, speedSum = 0, sizeSum = 0;

                // Process Agents
                for (let i = this.agents.length - 1; i >= 0; i--) {
                    let agent = this.agents[i];
                    
                    let nearby = this.grid.getNearby(agent.pos, agent.dna.perception);
                    agent.behavior(nearby);
                    agent.update();
                    agent.eat(this.resources);

                    if (!agent.active) {
                        this.stats.deaths++;
                        // Drop meat proportional to size/energy
                        let meatEnergy = (agent.dna.size * 20) + (agent.energy * 0.2);
                        this.resources.push(new Resource(agent.pos.x, agent.pos.y, ENTITY_MEAT, Math.min(meatEnergy, 150)));
                        this.agents.splice(i, 1);
                        if (this.inspectedAgent === agent) this.inspectedAgent = null; // Clear if dead
                        continue;
                    }

                    // Reproduction
                    if (agent.energy > agent.reproduceThreshold && this.agents.length < 1000) {
                        // Pay the reproductive cost
                        agent.energy *= 0.5;
                        
                        // Deep copy DNA and mutate
                        let childDNA = { ...agent.dna };
                        let child = new Agent(agent.pos.x, agent.pos.y, childDNA, agent.lineage, agent.generation + 1);
                        child.mutate();
                        
                        this.agents.push(child);
                        this.stats.births++;
                        if (child.generation > this.stats.maxGen) this.stats.maxGen = child.generation;
                    }

                    dietSum += agent.dna.diet;
                    speedSum += agent.dna.speed;
                    sizeSum += agent.dna.size;
                }

                // Clean resources
                this.resources = this.resources.filter(r => r.active && r.energy > 0.5);
                
                // Meat rots (slower than plant generation)
                for (let r of this.resources) {
                    if (r.type === ENTITY_MEAT) r.energy -= 0.02;
                    if (r.type === ENTITY_PLANT) plantCount++;
                    else meatCount++;
                }

                // Update HUD
                document.getElementById('hud-pop').innerText = this.agents.length;
                document.getElementById('hud-gen').innerText = this.stats.maxGen;
                document.getElementById('hud-plants').innerText = plantCount;
                document.getElementById('hud-meat').innerText = meatCount;
                document.getElementById('stat-births').innerText = this.stats.births;
                document.getElementById('stat-deaths').innerText = this.stats.deaths;
                
                if (this.agents.length > 0) {
                    document.getElementById('stat-speed').innerText = (speedSum / this.agents.length).toFixed(1);
                    document.getElementById('stat-size').innerText = (sizeSum / this.agents.length).toFixed(1);
                    let avgDiet = dietSum / this.agents.length;
                    document.getElementById('stat-diet').innerText = avgDiet < 0.3 ? "Herbivore" : (avgDiet > 0.7 ? "Carnivore" : "Omnivore");
                }
                
                this.updateInspectorUI();
            }

            render() {
                // Background & Env Map
                this.ctx.fillStyle = '#0a0b10';
                this.ctx.fillRect(0, 0, this.width, this.height);

                if (this.config.drawEnv) {
                    let grad = this.ctx.createRadialGradient(this.width/2, this.height/2, 0, this.width/2, this.height/2, Math.max(this.width, this.height)/2);
                    grad.addColorStop(0, 'rgba(74, 222, 128, 0.05)'); // Fertile center
                    grad.addColorStop(1, 'rgba(248, 113, 113, 0.1)'); // Harsh edge
                    this.ctx.fillStyle = grad;
                    this.ctx.fillRect(0, 0, this.width, this.height);
                }

                // Entities
                for (let r of this.resources) r.render(this.ctx);
                for (let a of this.agents) a.render(this.ctx);
            }

            loop() {
                if (!this.isRunning) return;
                for(let i=0; i<this.config.speed; i++) this.step();
                this.render();
                requestAnimationFrame(() => this.loop());
            }

            handleClick(e) {
                let rect = this.canvas.getBoundingClientRect();
                let x = (e.clientX || e.pageX) - rect.left;
                let y = (e.clientY || e.pageY) - rect.top;
                let clickPos = new Vector(x, y);
                
                // Find closest agent
                let closest = null, minDist = 400; // 20px radius
                for(let a of this.agents) {
                    let d = Vector.distSq(a.pos, clickPos);
                    if(d < minDist) { minDist = d; closest = a; }
                }

                if (closest) {
                    this.inspectedAgent = closest;
                    document.getElementById('inspector-panel').classList.add('open');
                    this.updateInspectorUI(true); // Force immediate update
                } else {
                    this.inspectedAgent = null;
                    document.getElementById('inspector-panel').classList.remove('open');
                }
                this.render(); // Force redraw to show selection
            }

            updateInspectorUI(force = false) {
                if (!this.inspectedAgent) return;
                let a = this.inspectedAgent;
                
                document.getElementById('insp-lineage').innerText = a.lineage.name;
                document.getElementById('insp-lineage').style.color = a.lineage.color;
                document.getElementById('insp-gen').innerText = a.generation;
                document.getElementById('insp-age').innerText = a.age;
                document.getElementById('insp-state').innerText = a.state;
                
                let pctE = (a.energy / a.maxEnergy) * 100;
                document.getElementById('insp-energy-val').innerText = `${Math.floor(a.energy)} / ${Math.floor(a.maxEnergy)}`;
                document.getElementById('insp-energy-bar').style.width = `${pctE}%`;
                
                // Traits
                if (force || this.ticks % 15 === 0) {
                    document.getElementById('insp-diet-val').innerText = a.dna.diet.toFixed(2);
                    document.getElementById('insp-diet-bar').style.width = `${a.dna.diet * 100}%`;
                    
                    document.getElementById('insp-size-val').innerText = a.dna.size.toFixed(1);
                    document.getElementById('insp-size-bar').style.width = `${(a.dna.size/10)*100}%`;
                    
                    document.getElementById('insp-speed-val').innerText = a.dna.speed.toFixed(1);
                    document.getElementById('insp-speed-bar').style.width = `${(a.dna.speed/5)*100}%`;
                    
                    let socPct = Math.max(0, Math.min(100, ((a.dna.sociability + 0.5) / 2) * 100));
                    document.getElementById('insp-soc-val').innerText = a.dna.sociability.toFixed(2);
                    document.getElementById('insp-soc-bar').style.width = `${socPct}%`;
                    
                    document.getElementById('insp-exp-val').innerText = a.dna.exploration.toFixed(2);
                    document.getElementById('insp-exp-bar').style.width = `${a.dna.exploration*100}%`;

                    // Generate Human Summary
                    let desc = `This <strong>${a.dna.size > 6 ? 'huge' : a.dna.size < 4 ? 'tiny' : 'average-sized'}</strong> organism belongs to the <strong>${a.lineage.name}</strong> lineage. `;
                    
                    if (a.dna.diet < 0.2) desc += `It is a strict <strong>herbivore</strong>, efficiently converting plants to energy. `;
                    else if (a.dna.diet > 0.8) desc += `It is a specialized <strong>scavenger/carnivore</strong>, relying on the remains of others. `;
                    else desc += `It is an <strong>omnivore</strong>, trading digestive efficiency for dietary flexibility. `;
                    
                    if (a.dna.sociability > 0.8) desc += `Highly social, it actively forms swarms for safety. `;
                    else if (a.dna.sociability < 0) desc += `Fiercely independent, it actively avoids others to prevent competition. `;

                    if (a.dna.speed > 3) desc += `Its high speed allows it to reach food quickly, but causes a rapid metabolic drain.`;
                    
                    document.getElementById('insp-summary').innerHTML = desc;
                }
            }

            bindUI() {
                document.getElementById('btn-toggle-controls').addEventListener('click', () => {
                    document.getElementById('controls-sidebar').classList.toggle('open');
                });
                
                document.getElementById('btn-close-inspector').addEventListener('click', () => {
                    this.inspectedAgent = null;
                    document.getElementById('inspector-panel').classList.remove('open');
                });

                document.getElementById('btn-pause').addEventListener('click', (e) => {
                    this.isRunning = !this.isRunning;
                    e.target.innerText = this.isRunning ? "Pause" : "Resume";
                    if (this.isRunning) this.loop();
                });

                document.getElementById('inp-speed').addEventListener('input', (e) => {
                    this.config.speed = parseInt(e.target.value);
                    document.getElementById('lbl-speed').innerText = `${this.config.speed}x`;
                });

                document.getElementById('btn-random').addEventListener('click', () => {
                    const seed = Math.random().toString(36).substring(2, 8).toUpperCase();
                    this.init(seed);
                });

                document.getElementById('btn-restart').addEventListener('click', () => {
                    let seed = document.getElementById('inp-seed').value || 'GENESIS';
                    this.init(seed);
                });

                document.getElementById('chk-env').addEventListener('change', e => this.config.drawEnv = e.target.checked);
                document.getElementById('chk-links').addEventListener('change', e => this.config.drawLinks = e.target.checked);
                document.getElementById('chk-lineage').addEventListener('change', e => this.config.drawLineage = e.target.checked);

                // Modals
                document.getElementById('btn-about').addEventListener('click', () => {
                    document.getElementById('modal-about').classList.add('open');
                });
                document.getElementById('btn-close-modal').addEventListener('click', () => {
                    document.getElementById('modal-about').classList.remove('open');
                });
            }
        }
