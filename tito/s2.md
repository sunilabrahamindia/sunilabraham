<p>This experiment models individual ants using local movement rules. Each ant senses nearby trail pheromone, deposits its own trail, responds to neighbours and searches for food. Under some conditions, positive feedback can reinforce a circular procession. The model is illustrative: it does not claim that every natural ant mill forms by exactly these mechanisms.</p>

    function RNG(seed){this.s=(seed>>>0)||1;}
    RNG.prototype.next=function(){this.s=(Math.imul(this.s,1664525)+1013904223)>>>0;return this.s/4294967296;};

    const wrap=(v,max)=>(v%max+max)%max;

    function resize(){

      state.width=Math.max(320,rect.width);state.height=Math.max(360,rect.height);

    }
    function buildObstacles(){

          {x:state.width*.34,y:state.height*.3,r:Math.min(34,state.width*.045)},
          {x:state.width*.66,y:state.height*.68,r:Math.min(42,state.width*.055)},
          {x:state.width*.51,y:state.height*.48,r:Math.min(23,state.width*.03)}

    }
    function idx(x,y){return wrap(Math.floor(x/state.width*state.cols),state.cols)+wrap(Math.floor(y/state.height*state.rows),state.rows)*state.cols;}

    function deposit(x,y,amount){state.field[idx(x,y)]=Math.min(1,state.field[idx(x,y)]+amount);}
    function isBlocked(x,y){return state.obstacles.some(o=>{const dx=Math.min(Math.abs(x-o.x),state.width-Math.abs(x-o.x)),dy=Math.min(Math.abs(y-o.y),state.height-Math.abs(y-o.y));return dx*dx+dy*dy<o.r*o.r;});}
    function addFood(x,y,amount=100){state.foods.push({x:wrap(x,state.width),y:wrap(y,state.height),amount});}
    function reset(seed){
      state.seed=seed>>>0||1;state.rng=new RNG(state.seed);state.tick=0;state.foodCollected=0;state.loopFrames=0;state.loopDetected=false;state.circularity=0;state.meanTrail=0;state.ants=[];state.foods=[];state.paths=[];state.field.fill(0);buildObstacles();
      const p=params();
      for(let i=0;i<p.population;i++){
        const x=rand(state.width*.08,state.width*.92),y=rand(state.height*.08,state.height*.92);
        state.ants.push({x,y,angle:rand(0,Math.PI*2),speed:rand(0.65,1.35),energy:rand(60,100),carrying:false,turn:rand(-1,1),trail:[],lastAngle:0});
      }
      if(p.environment==="food"){
        addFood(state.width*.2,state.height*.25,200);
        addFood(state.width*.78,state.height*.28,200);

    }
    function seedCircle(){
      const cx=state.width/2,cy=state.height/2,r=Math.min(state.width,state.height)*.22;
      state.field.fill(0);state.loopFrames=0;state.loopDetected=false;
      state.ants.forEach((a,i)=>{
        const t=(i/state.ants.length)*Math.PI*2;
        a.x=cx+Math.cos(t)*r+rand(-3,3);a.y=cy+Math.sin(t)*r+rand(-3,3);
        a.angle=t+Math.PI/2+rand(-.14,.14);a.energy=80;a.carrying=false;a.trail=[];
      });
      for(let t=0;t<Math.PI*2;t+=.025)deposit(cx+Math.cos(t)*r,cy+Math.sin(t)*r,.8);
      state.center={x:cx,y:cy};updateStatus("Circular trail seeded");render();
    }
    function updateStatus(message){$("am-status").textContent=message;}
    function updateControls(){
      $("am-population-value").textContent=$("am-population").value;
      $("am-following-value").textContent=$("am-following").value+"%";
      $("am-randomness-value").textContent=$("am-randomness").value+"%";
      $("am-evaporation-value").textContent=$("am-evaporation").value+"%";
    }
    function moveAnt(a,p){
      const senseDist=7,turnAngle=.62,follow=p.following;
      const left=sample(a.x+Math.cos(a.angle-turnAngle)*senseDist,a.y+Math.sin(a.angle-turnAngle)*senseDist);
      const ahead=sample(a.x+Math.cos(a.angle)*senseDist,a.y+Math.sin(a.angle)*senseDist);
      const right=sample(a.x+Math.cos(a.angle+turnAngle)*senseDist,a.y+Math.sin(a.angle+turnAngle)*senseDist);
      let steer=0;
      if(left>ahead&&left>right)steer=-turnAngle;
      else if(right>ahead&&right>left)steer=turnAngle;
      else if(Math.abs(left-right)<.012)steer=rand(-.16,.16);
      a.angle+=steer*follow;
      // Nearby ants provide a modest directional-alignment cue, not centralised control.
      let sx=0,sy=0,n=0;
      for(let j=0;j<state.ants.length;j+=3){
        const b=state.ants[j];if(b===a)continue;
        let dx=b.x-a.x,dy=b.y-a.y;
        if(Math.abs(dx)>state.width/2)dx-=Math.sign(dx)*state.width;
        if(Math.abs(dy)>state.height/2)dy-=Math.sign(dy)*state.height;
        if(dx*dx+dy*dy<20*20){sx+=Math.cos(b.angle);sy+=Math.sin(b.angle);n++;}
      }
      if(n>0&&state.rng.next()<.24*follow)a.angle+=angleDiff(Math.atan2(sy,sx),a.angle)*.12;
      // Food-seeking is stronger when not carrying food.
      if(p.environment==="food"&&!a.carrying){
        let target=null,best=Infinity;
        for(const f of state.foods){if(f.amount<=0)continue;let dx=f.x-a.x,dy=f.y-a.y;if(Math.abs(dx)>state.width/2)dx-=Math.sign(dx)*state.width;if(Math.abs(dy)>state.height/2)dy-=Math.sign(dy)*state.height;const d=dx*dx+dy*dy;if(d<best){best=d;target={x:dx,y:dy};}}
        if(target&&best<Math.min(state.width,state.height)*Math.min(state.width,state.height)*.16&&state.rng.next()<.65){
          const targetAngle=Math.atan2(target.y,target.x);a.angle+=angleDiff(targetAngle,a.angle)*.035;
        }
      }
      a.angle+=rand(-p.randomness*.72,p.randomness*.72);
      let nx=a.x+Math.cos(a.angle)*a.speed,ny=a.y+Math.sin(a.angle)*a.speed;
      if(isBlocked(nx,ny)){a.angle+=Math.PI*(.55+state.rng.next()*.9);nx=a.x+Math.cos(a.angle)*a.speed;ny=a.y+Math.sin(a.angle)*a.speed;}
      a.x=wrap(nx,state.width);a.y=wrap(ny,state.height);
      deposit(a.x,a.y,a.carrying?.055:.035);
      a.energy-=.012;
      if(p.environment==="food"){
        for(const f of state.foods){
          const dx=Math.min(Math.abs(f.x-a.x),state.width-Math.abs(f.x-a.x)),dy=Math.min(Math.abs(f.y-a.y),state.height-Math.abs(f.y-a.y));
          if(f.amount>0&&dx*dx+dy*dy<25){f.amount-=1;a.carrying=true;a.energy=Math.min(100,a.energy+2);state.foodCollected++;a.angle+=Math.PI+rand(-.35,.35);deposit(a.x,a.y,.35);break;}
        }
      }
      if(a.energy<=0){a.energy=100;a.x=rand(0,state.width);a.y=rand(0,state.height);a.angle=rand(0,Math.PI*2);a.carrying=false;}
      a.trail.push({x:a.x,y:a.y});if(a.trail.length>18)a.trail.shift();
    }
    function diffuse(p){
      const old=state.field,next=new Float32Array(old.length),decay=1-p.evaporation*.012;
      for(let y=0;y<state.rows;y++)for(let x=0;x<state.cols;x++){
        const i=x+y*state.cols,l=wrap(x-1,state.cols)+y*state.cols,r=wrap(x+1,state.cols)+y*state.cols,u=x+wrap(y-1,state.rows)*state.cols,d=x+wrap(y+1,state.rows)*state.cols;
        next[i]=(old[i]*.76+(old[l]+old[r]+old[u]+old[d])*.06)*decay;
      }
      state.field=next;
    }
    function detectLoop(){
      if(state.ants.length<8){state.circularity=0;state.loopFrames=0;state.loopDetected=false;return;}
      let mx=0,my=0;
      // Circular mean centre avoids false jumps at wrapped edges in most ordinary configurations.
      let cx=0,cy=0;
      for(const a of state.ants){cx+=Math.cos(a.x/state.width*Math.PI*2);cy+=Math.sin(a.x/state.width*Math.PI*2);mx+=Math.cos(a.y/state.height*Math.PI*2);my+=Math.sin(a.y/state.height*Math.PI*2);}
      const centre={x:(Math.atan2(cy,cx)/(Math.PI*2)+1)%1*state.width,y:(Math.atan2(my,mx)/(Math.PI*2)+1)%1*state.height};
      let tangential=0,radial=0,count=0;
      for(const a of state.ants){
        let dx=a.x-centre.x,dy=a.y-centre.y;if(Math.abs(dx)>state.width/2)dx-=Math.sign(dx)*state.width;if(Math.abs(dy)>state.height/2)dy-=Math.sign(dy)*state.height;
        const r=Math.hypot(dx,dy);if(r<12)continue;
        const radialVel=(Math.cos(a.angle)*dx+Math.sin(a.angle)*dy)/r;
        const tangentVel=(-Math.sin(a.angle)*dx+Math.cos(a.angle)*dy)/r;
        radial+=Math.abs(radialVel);tangential+=Math.abs(tangentVel);count++;
      }
      const consistency=count?Math.abs(tangential-radial)/(tangential+radial+1e-6):0;
      // High consistency alone is insufficient: require many ants, sustained rotation and a coherent annulus.
      const radii=state.ants.map(a=>{let dx=a.x-centre.x,dy=a.y-centre.y;if(Math.abs(dx)>state.width/2)dx-=Math.sign(dx)*state.width;if(Math.abs(dy)>state.height/2)dy-=Math.sign(dy)*state.height;return Math.hypot(dx,dy);}).sort((a,b)=>a-b);
      const spread=radii[Math.floor(radii.length*.8)]-radii[Math.floor(radii.length*.2)];
      const annulus=1-Math.min(1,spread/(Math.min(state.width,state.height)*.42));
      state.circularity=Math.max(0,Math.min(1,consistency*.55+annulus*.45));
      const qualifies=state.circularity>.68&&count>state.ants.length*.65;
      state.loopFrames=qualifies?state.loopFrames+1:Math.max(0,state.loopFrames-2);
      state.loopDetected=state.loopFrames>180;
      state.center=centre;
    }
    function step(){
      const p=params();
      if(state.field.length===0)return;
      for(let i=0;i<p.speed;i++){
        state.tick++;
        for(const a of state.ants)moveAnt(a,p);
        diffuse(p);
        if(state.tick%8===0)detectLoop();
        if(state.tick%6===0&&p.environment==="food"&&state.foods.length<12){
          state.foods=state.foods.filter(f=>f.amount>0);
          if(state.foods.length<8)addFood(rand(state.width*.08,state.width*.92),rand(state.height*.08,state.height*.92),100);
        }
      }
      state.meanTrail=state.field.reduce((s,v)=>s+v,0)/Math.max(1,state.field.length);
      $("am-pop").textContent=state.ants.length;

## How to run an experiment

1. Start with the default population of 300 ants and the default settings.
2. Observe whether the trail field strengthens into a circular pattern without seeding one.
3. Select **Seed circular trail** to compare an imposed circle with spontaneous movement.
4. Increase randomness or evaporation, then use **Disturb the trail** to see whether the pattern persists.
5. Change the environment to obstacles and barriers to observe how the agents alter their routes.

The circularity score is a heuristic based on the distribution and direction of movement. A high score is not, by itself, proof of a biologically realistic ant mill.
