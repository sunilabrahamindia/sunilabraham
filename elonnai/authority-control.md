This page documents authority control identifiers, bibliographic records, scholarly profiles, institutional affiliations, conference records, and related metadata associated with [Elonnai Hickok](/elonnai/). It is intended to support long-term discoverability, citation consistency, digital preservation, and interoperability across research databases, institutional repositories, conference archives, policy organisations, and web-based scholarly infrastructure.

The page consolidates external records connected with Elonnai Hickok's work in technology policy, privacy, cybersecurity, surveillance, digital identity, internet governance, intermediary liability, and artificial intelligence governance, as well as her research, institutional affiliations, publications, conference participation, and policy work.

As metadata systems frequently fragment identities across variant name formats, institutional affiliations, publication databases, and web-based records, this page also serves as a reconciliation layer for authority control and bibliographic standardisation within The Sunil Abraham Project.

<script>
(function () {
  var searchInput = document.getElementById('authority-search');
  var clearBtn    = document.getElementById('ac-clear');
  var countEl     = document.getElementById('ac-count');
  var emptyEl     = document.getElementById('ac-empty');
  var items       = document.querySelectorAll('.ac-item');
  var groups      = document.querySelectorAll('.ac-group-card');
  var TOTAL       = items.length;

  function writeURL(q) {
    var currentParams = new URLSearchParams(window.location.search);
    var darkmodeValue = currentParams.get('darkmode');
    
    var params = new URLSearchParams();
    if (q) params.set('q', q);
    if (darkmodeValue !== null) params.set('darkmode', darkmodeValue);

    var newURL = params.toString()
      ? window.location.pathname + '?' + params.toString()
      : window.location.pathname;
    history.replaceState(null, '', newURL);
  }

  function readURL() {
    var params = new URLSearchParams(window.location.search);
    var q = params.get('q');
    if (q) {
      searchInput.value = q;
      runSearch(q.toLowerCase().trim(), false);
    }
  }

  function updateCount(n) {
    countEl.textContent = n + (n === 1 ? ' record' : ' records');
  }

  function runSearch(q, shouldUpdateURL) {
    if (typeof shouldUpdateURL === 'undefined') shouldUpdateURL = true;
    var visibleTotal = 0;

    items.forEach(function (item) {
      var label = (item.dataset.label || '').toLowerCase();
      var value = (item.dataset.value || '').toLowerCase();
      var text  = item.textContent.toLowerCase();
      var match = !q || label.includes(q) || value.includes(q) || text.includes(q);
      item.setAttribute('aria-hidden', match ? 'false' : 'true');
      if (match) visibleTotal++;
    });

    groups.forEach(function (grp) {
      var groupItems = grp.querySelectorAll('.ac-item');
      var visibleInGroup = 0;
      groupItems.forEach(function (it) {
        if (it.getAttribute('aria-hidden') !== 'true') {
          visibleInGroup++;
        }
      });

      grp.setAttribute('aria-hidden', visibleInGroup > 0 ? 'false' : 'true');

      var badge = grp.querySelector('.ac-group-badge');
      if (badge) {
        badge.textContent = visibleInGroup;
      }
    });

    updateCount(q ? visibleTotal : TOTAL);
    emptyEl.classList.toggle('visible', visibleTotal === 0 && !!q);
    clearBtn.classList.toggle('visible', !!q);
    
    if (shouldUpdateURL) {
      writeURL(q);
    }
  }

  searchInput.addEventListener('input', function () {
    runSearch(this.value.toLowerCase().trim());
  });

  clearBtn.addEventListener('click', function () {
    searchInput.value = '';
    runSearch('');
    searchInput.focus();
  });

  searchInput.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && this.value) {
      this.value = '';
      runSearch('');
    }
  });

  updateCount(TOTAL);
  readURL();
})();
</script>
